# .bashrc becomes a template too, with a conditional block:
# one alias exists only where the data says machine=laptop
# (edited template kept as examples/dot_bashrc.tmpl)

$ cat .local/share/chezmoi/dot_bashrc.tmpl
# Alex's .bashrc -- the shared starting point on every machine.

# History: kept, and shared sensibly between shells.
HISTSIZE=5000
HISTFILESIZE=2000

# Prompt: user@host and the working directory.
PS1='\u@\h:\w\$ '

# Aliases everyone gets.
alias ll='ls -alF'
alias gs='git status'
alias gd='git diff'
{{ if eq .machine "laptop" -}}

# Laptop only: the projects directory shortcut.
alias proj='cd ~/projects'
{{ end -}}
[exit 0]

# On the laptop the block renders in

$ chezmoi apply
[exit 0]
