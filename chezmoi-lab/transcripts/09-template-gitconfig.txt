# Previously: .gitconfig was re-added with --template, so the
# source file is dot_gitconfig.tmpl, committed as a rename.
# Now Alex edits the template so two answers come from data
# instead of being baked in. The edited template:

$ cat .local/share/chezmoi/dot_gitconfig.tmpl
[user]
	name = Alex Mercer
	email = {{ .email }}
[core]
	editor = {{ .editor }}
[init]
	defaultBranch = main
[alias]
	st = status
	co = checkout
[exit 0]

# The data itself lives in the config file, which is
# per-machine and never enters the repository
# (this one is kept as examples/chezmoi.toml.laptop)

$ cat .config/chezmoi/chezmoi.toml
# Alex's config on the laptop: ~/.config/chezmoi/chezmoi.toml
# This file is NOT in the dotfiles repository. It is written on
# each machine, by hand, and it is how one shared set of templates
# renders differently per machine: templates read these values.

[data]
    machine = "laptop"
    email = "alex@example.com"
    editor = "vim"
[exit 0]

# Apply on the laptop: the template renders with the laptop's
# answers, which match what was already there -- no change

$ chezmoi apply
[exit 0]
